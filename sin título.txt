    <!-- JavaScript: Generador de Lluvia de Flores -->
    <script>
        // Configuración
        const totalFlowers = 50; // Número de flores cayendo a la vez
        const symbols = ['🌻', '🌼', '✨', '💛', 'pétalo']; // Lo que puede caer

        function createFlower() {
            const flower = document.createElement('div');
            flower.classList.add('flower');

            // Seleccionar símbolo aleatorio
            const symbol = symbols[Math.floor(Math.random() * symbols.length)];
            
            // Si es un pétalo, le damos estilo especial, si no, el emoji
            if (symbol === 'pétalo') {
                flower.style.width = '15px';
                flower.style.height = '15px';
                flower.style.backgroundColor = '#FFD700'; /* Amarillo Oro */
                flower.style.borderRadius = '50% 0 50% 50%'; /* Forma de pétalo */
                flower.style.opacity = '0.7';
            } else {
                flower.textContent = symbol;
            }

            // Posición horizontal aleatoria (0% a 100% del ancho)
            flower.style.left = Math.random() * 100 + 'vw';

            // Tamaño aleatorio (entre 1em y 2.5em)
            const size = Math.random() * 1.5 + 1;
            flower.style.fontSize = size + 'em';

            // Duración de caída aleatoria (entre 5s y 15s) para variar velocidad
            const duration = Math.random() * 10 + 5;
            flower.style.animationDuration = duration + 's';

            // Retraso de inicio aleatorio para que no caigan todas a la vez
            flower.style.animationDelay = Math.random() * 5 + 's';

            document.body.appendChild(flower);

            // Eliminar el elemento después de que termine la animación para no saturar
            setTimeout(() => {
                flower.remove();
            }, (duration + 5) * 1000); // Duración + delay
        }

        // Crear las flores iniciales
        for (let i = 0; i < totalFlowers; i++) {
            createFlower();
        }

        // Crear nuevas flores continuamente
        setInterval(createFlower, 300); // Crea una flor nueva cada 300ms
    </script>
